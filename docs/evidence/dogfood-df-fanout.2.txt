import pytest
from inventory import stock


def test_remove_partial_leaves_remainder():
    s = stock.Stock()
    s.add("A", 5)
    s.remove("A", 3)
    assert s.level("A") == 2


def test_remove_exact_level_leaves_zero():
    s = stock.Stock()
    s.add("A", 5)
    s.remove("A", 5)
    assert s.level("A") == 0


def test_remove_too_much_raises_and_keeps_level():
    s = stock.Stock()
    s.add("A", 5)
    with pytest.raises(stock.InsufficientStock):
        s.remove("A", 6)
    assert s.level("A") == 5


def test_remove_unknown_sku_raises():
    s = stock.Stock()
    with pytest.raises(stock.InsufficientStock):
        s.remove("NEW", 1)
    assert s.level("NEW") == 0


def test_insufficient_stock_is_exception():
    assert issubclass(stock.InsufficientStock, Exception)
EOF
