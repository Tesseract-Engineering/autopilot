import pytest
from inventory import stock as stock_mod


def test_remove_more_than_level_raises_and_keeps_level():
    s = stock_mod.Stock()
    s.add("apple", 5)
    with pytest.raises(stock_mod.InsufficientStock):
        s.remove("apple", 6)
    assert s.level("apple") == 5


def test_remove_exact_level_gives_zero():
    s = stock_mod.Stock()
    s.add("apple", 5)
    s.remove("apple", 5)
    assert s.level("apple") == 0


def test_remove_partial():
    s = stock_mod.Stock()
    s.add("apple", 5)
    s.remove("apple", 2)
    assert s.level("apple") == 3


def test_remove_unknown_sku_raises():
    s = stock_mod.Stock()
    with pytest.raises(stock_mod.InsufficientStock):
        s.remove("pear", 1)
    assert s.level("pear") == 0


def test_insufficient_stock_is_exception():
    assert issubclass(stock_mod.InsufficientStock, Exception)
EOF
